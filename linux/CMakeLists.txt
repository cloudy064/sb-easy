cmake_minimum_required(VERSION 3.24)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../config/client-releases.json" client_releases)
string(JSON desktop_version GET "${client_releases}" development linux version)
project(sbeasy_linux_desktop VERSION "${desktop_version}" LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
include(CTest)
include(GNUInstallDirs)
find_package(PkgConfig REQUIRED)
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
find_package(CURL 7.85 REQUIRED)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0>=3.24)
pkg_check_modules(SECRET REQUIRED IMPORTED_TARGET libsecret-1>=0.20)
pkg_check_modules(POLKIT REQUIRED IMPORTED_TARGET polkit-gobject-1)
pkg_check_modules(GIO_UNIX REQUIRED IMPORTED_TARGET gio-unix-2.0>=2.66)
if(NOT SB_EASY_SINGBOX_BRIDGE_LIBRARY OR NOT SB_EASY_SINGBOX_BRIDGE_INCLUDE_DIR)
    message(FATAL_ERROR "Run linux/scripts/prepare-core.sh and pass its archive/include directory")
endif()
file(STRINGS "${SB_EASY_SINGBOX_BRIDGE_INCLUDE_DIR}/libsb_easy_singbox.h" bridge_cancel REGEX SBEasySingBoxCancelStart)
if(NOT bridge_cancel)
    message(FATAL_ERROR "The embedded bridge must support cancellable startup; rebuild with linux/scripts/prepare-core.sh")
endif()
set(SB_EASY_EMBED_SINGBOX ON CACHE BOOL "Embedded pinned runtime" FORCE)
add_subdirectory(../c core EXCLUDE_FROM_ALL)
# These limits apply only to this desktop's private sb_core build.
target_compile_definitions(sb_core PRIVATE AGENT_MAX_BODY=2097152
    YAML_MAX_DEPTH=64 YAML_MAX_NODES=20000 YAML_CONVERT_BUDGET=30000 YAML_CONVERT_DEPTH=64
    YAML_CONVERT_STRING_BUDGET=2097152 YAML_MAX_FIX_BYTES=2097152 YAML_MAX_FIX_ROUNDS=32 YAML_REJECT_LIMIT_EXHAUSTION=1)
add_library(sbl_desktop STATIC src/common.c src/vault.c src/profile.c src/config.c
    src/runtime.c src/system_proxy.c ../apple/Core/core.c)
target_include_directories(sbl_desktop PUBLIC src ../apple/Core/include)
target_link_libraries(sbl_desktop PUBLIC sb_core PkgConfig::GTK PkgConfig::SECRET)
target_compile_definitions(sbl_desktop PUBLIC SBL_VERSION="${desktop_version}"
    PRIVATE SBA_PLATFORM_PREFIX="sb-easy-linux-" SBA_PLATFORM_LABEL="Linux" SBA_ALLOW_PROCESS_RULES=1)
target_compile_options(sbl_desktop PRIVATE -Wall -Wextra -Wshadow -Werror)
add_library(sbl_controller STATIC src/controller.c)
target_link_libraries(sbl_controller PUBLIC sbl_desktop)
target_compile_options(sbl_controller PRIVATE -Wall -Wextra -Wshadow -Werror)
add_executable(sb-easy-desktop src/app.c)
target_link_libraries(sb-easy-desktop PRIVATE sbl_controller PkgConfig::GIO_UNIX)
target_compile_options(sb-easy-desktop PRIVATE -Wall -Wextra -Wshadow -Werror)
add_executable(sb-easy-tunnel src/tunnel_service.c)
target_link_libraries(sb-easy-tunnel PRIVATE sbl_desktop PkgConfig::POLKIT)
target_compile_options(sb-easy-tunnel PRIVATE -Wall -Wextra -Wshadow -Werror)

if(BUILD_TESTING)
    foreach(name core service keyring system_proxy tun)
        add_executable(linux_${name}_test tests/${name}_test.c)
        target_link_libraries(linux_${name}_test PRIVATE sbl_desktop)
        target_compile_options(linux_${name}_test PRIVATE -Wall -Wextra -Wshadow -Werror)
    endforeach()
    add_executable(linux_controller_test tests/controller_test.c)
    target_link_libraries(linux_controller_test PRIVATE sbl_controller)
    target_compile_options(linux_controller_test PRIVATE -Wall -Wextra -Wshadow -Werror)
    add_executable(linux_ui_smoke src/app.c)
    target_link_libraries(linux_ui_smoke PRIVATE sbl_controller PkgConfig::GIO_UNIX)
    target_compile_definitions(linux_ui_smoke PRIVATE SBL_UI_TEST=1)
    target_compile_options(linux_ui_smoke PRIVATE -Wall -Wextra -Wshadow -Werror)
    add_test(NAME linux_core_contract COMMAND linux_core_test)
    add_test(NAME linux_live_runtime COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/live_fixture.py $<TARGET_FILE:linux_core_test>)
    add_test(NAME linux_private_bus COMMAND dbus-run-session -- ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/service_fixture.py $<TARGET_FILE:sb-easy-tunnel> $<TARGET_FILE:linux_service_test>)
    add_test(NAME linux_keyring COMMAND dbus-run-session -- ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/keyring_fixture.py $<TARGET_FILE:linux_keyring_test>)
    add_test(NAME linux_system_proxy COMMAND linux_system_proxy_test)
    set_tests_properties(linux_system_proxy PROPERTIES ENVIRONMENT GSETTINGS_BACKEND=memory)
    add_test(NAME linux_native_gtk COMMAND xvfb-run -a -s "-screen 0 1440x1000x24" dbus-run-session -- ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/ui_fixture.py $<TARGET_FILE:linux_ui_smoke> ${CMAKE_CURRENT_BINARY_DIR}/ui-smoke.png)
    add_test(NAME linux_controller COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/controller_fixture.py $<TARGET_FILE:linux_controller_test>)
    set_tests_properties(linux_live_runtime linux_private_bus linux_keyring linux_native_gtk linux_controller PROPERTIES TIMEOUT 60)
endif()

install(TARGETS sb-easy-desktop RUNTIME DESTINATION bin)
install(TARGETS sb-easy-tunnel RUNTIME DESTINATION libexec)
install(FILES packaging/io.sbeasy.Desktop.desktop DESTINATION share/applications)
install(FILES ../assets/branding/sb-easy-mark-v2.svg DESTINATION share/icons/hicolor/scalable/apps RENAME io.sbeasy.Desktop.svg)
install(FILES packaging/io.sbeasy.Desktop.Tunnel.service DESTINATION share/dbus-1/system-services)
install(FILES packaging/io.sbeasy.Desktop.Tunnel.conf DESTINATION share/dbus-1/system.d)
install(FILES packaging/io.sbeasy.Desktop.policy DESTINATION share/polkit-1/actions)
install(FILES packaging/sb-easy-tunnel.service DESTINATION lib/systemd/system)
install(PROGRAMS scripts/configure-repository.py scripts/update.py DESTINATION libexec/sb-easy-desktop)
install(FILES README.md THIRD_PARTY_NOTICES.md DESTINATION share/doc/sb-easy-desktop)
if(EXISTS "${SB_EASY_SINGBOX_BRIDGE_INCLUDE_DIR}/provenance.json")
    install(FILES "${SB_EASY_SINGBOX_BRIDGE_INCLUDE_DIR}/provenance.json" DESTINATION share/doc/sb-easy-desktop RENAME core-provenance.json)
endif()
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
set(CPACK_PACKAGE_NAME sb-easy-desktop)
set(CPACK_PACKAGE_VENDOR sb-easy)
set(CPACK_PACKAGE_CONTACT "sb-easy contributors")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Native GTK desktop with independent and managed sing-box connections")
set(CPACK_PACKAGE_VERSION "${desktop_version}")
set(CPACK_DEBIAN_PACKAGE_SECTION net)
set(CPACK_DEBIAN_PACKAGE_PRIORITY optional)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "systemd, dbus, polkitd, python3 (>= 3.11), gnupg, apt (>= 2.6), ca-certificates")
set(CPACK_DEBIAN_PACKAGE_RECOMMENDS "gnome-keyring, policykit-1-gnome | polkit-kde-agent-1")
set(CPACK_DEBIAN_PACKAGE_CONTROL_EXTRA "${CMAKE_CURRENT_SOURCE_DIR}/packaging/postinst;${CMAKE_CURRENT_SOURCE_DIR}/packaging/prerm;${CMAKE_CURRENT_SOURCE_DIR}/packaging/postrm")
set(CPACK_DEBIAN_COMPRESSION_TYPE xz)
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64)$")
    set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE amd64)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
    set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE arm64)
else()
    message(FATAL_ERROR "Linux desktop package architecture has not been defined for this target")
endif()
find_program(DPKG_SHLIBDEPS dpkg-shlibdeps)
if(DPKG_SHLIBDEPS)
    set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
else()
    # A package from a non-Debian host is a local inspection artifact only.
    # Release packages are built/tested in packaging/Dockerfile, with shlibdeps.
    set(CPACK_DEBIAN_PACKAGE_DEPENDS "${CPACK_DEBIAN_PACKAGE_DEPENDS}, libc6, libgtk-3-0, libglib2.0-0, libsecret-1-0, libpolkit-gobject-1-0, libcurl4, libssl3, libsqlite3-0, libyaml-0-2")
endif()
include(CPack)
